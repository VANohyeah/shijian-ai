```ts
export default defineEventHandler(async (event) => {
  const body = await readBody<{ messages: any[] }>(event)
  const key = process.env.DEEPSEEK_API_KEY
  if (!key) throw createError({ statusCode: 500, statusMessage: 'DEEPSEEK_API_KEY 鏈厤缃? })

  const upstream = await fetch(`${process.env.DEEPSEEK_BASE_URL}/chat/completions`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${key}`,
    },
    body: JSON.stringify({
      model: process.env.DEEPSEEK_MODEL ?? 'deepseek-flash',
      messages: body.messages,
      stream: true,
      temperature: 0.3,
    }),
  })

  if (!upstream.ok || !upstream.body) {
    const detail = await upstream.text()
    throw createError({ statusCode: upstream.status, statusMessage: detail.slice(0, 300) })
  }

  setResponseHeaders(event, {
    'Content-Type': 'text/event-stream; charset=utf-8',
    'Cache-Control': 'no-cache, no-transform',
    Connection: 'keep-alive',
  })

  return sendStream(event, upstream.body as ReadableStream)
})
```
